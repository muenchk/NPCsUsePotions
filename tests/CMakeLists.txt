cmake_minimum_required(VERSION 3.22)
project(ActorRegistryTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(actor_registry_tests ActorRegistryTests.cpp)
target_compile_features(actor_registry_tests PRIVATE cxx_std_20)
target_include_directories(actor_registry_tests PRIVATE ../include)
target_link_libraries(actor_registry_tests PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(actor_registry_tests PRIVATE /W4 /WX /EHsc)
endif()
add_test(NAME actor_registry COMMAND actor_registry_tests)
set_tests_properties(actor_registry PROPERTIES TIMEOUT 30)
